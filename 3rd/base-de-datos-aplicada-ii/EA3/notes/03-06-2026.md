```sql
DECLARE
    -- variables de uso en la query
    v_nombre                    cliente.nombre_cli%TYPE;
    v_apellido_paterno          VARCHAR2(100);
    v_apellido_materno          VARCHAR2(100);
    v_rut                       cliente.numrut_cli%TYPE;
    v_dv                        cliente.dvrut_cli%TYPE;
    v_renta                     NUMERIC(10);
    v_id_estado_civil           cliente.id_estcivil%TYPE;
    
    v_estado_civil              estado_civil.desc_estcivil%TYPE;
    -- variable de uso en negocio
 
 
    v_sueldo_aumentado          NUMERIC(10);
BEGIN
    -- búsqueda de información
    SELECT
        nombre_cli
        ,appaterno_cli
        ,apmaterno_cli
        ,numrut_cli
        ,dvrut_cli
        ,renta_cli
        ,id_estcivil
    INTO
        v_nombre
        ,v_apellido_paterno
        ,v_apellido_materno
        ,v_rut
        ,v_dv
        ,v_renta
        ,v_id_estado_civil
    FROM cliente
    WHERE numrut_cli = 12467572
    ;
 
 
    SELECT
        desc_estcivil
    INTO
        v_estado_civil
    FROM estado_civil
    where id_estcivil = v_id_estado_civil
    ;
    -- lógica de negocio
    v_sueldo_aumentado := v_renta + v_renta *0.1;
    
    -- mensaje de salida
    DBMS_OUTPUT.PUT_LINE('DATOS DEL CLIENTE');
    DBMS_OUTPUT.PUT_LINE('-----------------');
    DBMS_OUTPUT.PUT_LINE('Nombre: '||v_nombre||' '||v_apellido_paterno||' '||v_apellido_materno );
    DBMS_OUTPUT.PUT_LINE('RUN: '||v_rut||'-'||v_dv);
    DBMS_OUTPUT.PUT_LINE('Estado Civil: '||v_estado_civil);
    DBMS_OUTPUT.PUT_LINE('Renta: '||TO_CHAR(v_renta,'$999G999G999'));
    DBMS_OUTPUT.PUT_LINE('Sueldo Aumentado: '||TO_CHAR(v_sueldo_aumentado,'$999G999G999'));
    DBMS_OUTPUT.PUT_LINE('FIN');
EXCEPTION 
    WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('Error');
END;
```
