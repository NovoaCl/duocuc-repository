SELECT
    EXTRACT(YEAR FROM SYSDATE) AS anio,
    e.id_emp,
    e.numrun_emp,
    e.dvrun_emp,
    e.pnombre_emp ||' '|| e.snombre_emp ||' '|| e.appaterno_emp ||' '|| e.amaterno_emp AS ,
    c.nombre_comuna,
    e.sueldo_base,
    TRUNC(e.sueldo_base / 100000) AS porcentaje,
    ROUND(e.sueldo_base * TRUNC(e.sueldo_base / 100000)/100) AS mov_normal, -- 0.2
    CASE e.id_comuna
        WHEN 117 THEN 20000
        WHEN 118 THEN 25000
        WHEN 119 THEN 30000
        WHEN 120 THEN 35000
        WHEN 121 THEN 40000
        ELSE 0
    END AS valor_movil_extra


FROM empleado e
    JOIN comuna c ON e.id_comuna = c.id_comuna
-- WHERE e.id_emp = 110
;

DECLARE
    -- variables para recorrer los registros
    v_id_emp_min        NUMERIC;
    v_id_emp_max        NUMERIC;
    v_cant_emp          NUMERIC:=0;   

    -- variables de la query
    v_anno          NUMERIC;
    v_id_emp        empleado.id_emp%TYPE;
    v_run           empleado.numrun_emp%TYPE;
    v_dv            empleado.dvrun_emp%TYPE;
    v_nombre        VARCHAR2(255);
    v_comuna        comuna.nombre_comuna%TYPE;
    v_sueldo_base   empleado.sueldo_base%TYPE;
    v_porcentaje    NUMERIC;
    v_mov_normal    NUMERIC;
    v_mov_extra     NUMERIC;
    
    -- variable de negocio
    v_total         NUMERIC;



BEGIN
    SELECT
    MIN(id_emp) AS MIN,
    MAX(id_emp)AS MAX,
    COUNT(*) AS CANTIDAD
    
    INTO
        v_id_emp_min,
        v_id_emp_max,
        v_cant_emp

    FROM empleado
    ;

    LOOP
    -- imprime los nombres
        DBMS_OUTPUT.put_line ('LOOP');
        v_id_emp_min := v_id_emp_max + 1;
        EXIT WHEN v_id_emp_min > v_id_emp_max;
    END LOOP

    DBMS_OUTPUT.put_line ('FIN');

END
;
