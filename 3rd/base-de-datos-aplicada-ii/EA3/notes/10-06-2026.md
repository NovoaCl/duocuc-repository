DECLARE
    -- variables de uso para cada columna de la query
    v_nro_cliente           cliente.nro_cliente%TYPE;
    v_run                   VARCHAR2(15);
    v_nombre_completo       VARCHAR2(250);
    v_tipo_cliente          tipo_cliente.nombre_tipo_cliente%TYPE;
    v_monto_solicitado      NUMERIC;

    -- variable de uso en negocio
    v_monto_pesos_todo_suma NUMERIC;
    v_pesos_adicionales     NUMERIC;

BEGIN
    -- búsqueda de información
    SELECT 
        c.nro_cliente,
        TO_CHAR(c.numrun,'999g999g999') ||'-'|| c.dbrun AS rut,
        pnombre ||' '|| snombre ||' '|| appaterno ||' '|| apmaterno AS nombre,
        tc.nombre_tipo_cliente,
        SUM(cc.monto_solicitado)
    
    INTO
    v_nro_cliente,
    v_run,
    v_nombre_completo,
    v_tipo_cliente,
    v_monto_solicitado

    FROM cliente c
        JOIN tipo_cliente tc 
            ON c.cod_tipo_cliente = tc.cod_tipo_cliente
        JOIN credito_cliente cc
            ON c.nro_cliente = cc.nro_cliente

    WHERE
        c.numrun = 21242003
    GROUP BY
        c.nro_cliente,
        TO_CHAR(c.numrun,'999g999g999') ||'-'|| c.dbrun,
        pnombre ||' '|| snombre ||' '|| appaterno ||' '|| apmaterno,
        tc.nombre_tipo_cliente
    ;

    -- Calculo de los pesos
    v_monto_pesos_todo_suma := ROUND(v_monto_solicitado / 100000 * 1200);

    -- validamos si el cliente es independiente o no

    IF v_tipo_cliente = 'Trabajadores independientes' THEN
        DBMS_OUTPUT.PUT_LINE('Es independiente');

        -- Pesos adicionales
        v_pesos_adicionales  := CASE 
            WHEN v_monto_solicitado < 1000000 THEN
                ROUND(v_monto_solicitado / 100000 * 1200);
        END;
    END IF
    ;

    DBMS_OUTPUT.PUT_LINE('Fin del proceso');

EXCEPTION
    WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('Error');

END
;

DECLARE
    --
    v_rut_consultar         NUMERIC := 21815820; --21242003 - 21815820;
    -- primero creamos una variable para cada columna de la query
    v_nro_cliente           cliente.nro_cliente%TYPE;
    v_run                   VARCHAR2(15);
    v_nombre_completo       VARCHAR2(255);
    v_tipo_cliente          tipo_cliente.nombre_tipo_cliente%TYPE;
    v_monto_solicitado      NUMERIC;
    -- variables para el negocio
    v_monto_pesos_todo_suma NUMERIC;
    v_pesos_adicionales     NUMERIC := 0;
BEGIN
    SELECT
        c.nro_cliente
        ,TO_CHAR(c.numrun, '999g999g999') ||'-'|| c.dvrun AS rut
        ,pnombre ||' '|| snombre ||' '|| appaterno ||' '|| apmaterno AS nombre
        ,tc.nombre_tipo_cliente
        ,SUM(cc.monto_solicitado) AS monto
    INTO
        v_nro_cliente
        ,v_run
        ,v_nombre_completo
        ,v_tipo_cliente
        ,v_monto_solicitado
    FROM cliente c
        INNER JOIN tipo_cliente tc
            ON c.cod_tipo_cliente = tc.cod_tipo_cliente
        JOIN credito_cliente cc
            ON c.nro_cliente = cc.nro_cliente
    WHERE 
        c.numrun = v_rut_consultar --21242003--21815820
    GROUP BY
        c.nro_cliente
        ,TO_CHAR(c.numrun, '999g999g999') ||'-'|| c.dvrun
        ,pnombre ||' '|| snombre ||' '|| appaterno ||' '|| apmaterno
        ,tc.nombre_tipo_cliente
    ;
    
    -- cálculo de los pesos
    v_monto_pesos_todo_suma := ROUND(v_monto_solicitado/100000 * 1200);
    DBMS_OUTPUT.put_line('pesos: '||v_monto_pesos_todo_suma);
    -- validamos si el cliente es independiente o no
    IF v_tipo_cliente = 'Trabajadores independientes' THEN
        -- pesos adicionales
        v_pesos_adicionales := CASE
            WHEN v_monto_solicitado < 1000000 
                THEN ROUND(v_monto_solicitado/100000 * 100)
            WHEN v_monto_solicitado BETWEEN 1000000 AND 3000000
                THEN ROUND(v_monto_solicitado/100000 * 300)
            WHEN v_monto_solicitado > 3000000
                THEN ROUND(v_monto_solicitado/100000 * 550) 
            ELSE 0
        END;
        
    END IF;   
    DBMS_OUTPUT.put_line('pesos adicional: '||v_pesos_adicionales);
    
    DBMS_OUTPUT.put_line('total: '||(v_monto_pesos_todo_suma+v_pesos_adicionales));
 
 
 
 
    DBMS_OUTPUT.put_line('Fin del proceso');
EXCEPTION 
    WHEN OTHERS THEN
    DBMS_OUTPUT.put_line('Error');
END;
