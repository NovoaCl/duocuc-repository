```sql

DECLARE
    -- MAESTRO: las sucursales que tienen al menos un credito.
    CURSOR c_sucursales IS
        SELECT DISTINCT s.id_sucursal, s.direccion
        FROM SUCURSAL_BANCO s
        JOIN CREDITO_CLIENTE cc ON cc.id_sucursal = s.id_sucursal
        ORDER BY s.id_sucursal;
    -- DETALLE: recibe por parametro la sucursal del maestro.
    CURSOR c_creditos (p_sucursal NUMBER) IS
        SELECT nro_solic_credito, monto_credito, total_cuotas_credito
        FROM CREDITO_CLIENTE
        WHERE id_sucursal = p_sucursal
        ORDER BY nro_solic_credito;

        v_subtotal NUMBER(12);
        v_total NUMBER(12) := 0;
        v_cuenta PLS_INTEGER;
BEGIN
    FOR v_suc IN c_sucursales LOOP -- ciclo externo
        DBMS_OUTPUT.PUT_LINE('SUCURSAL ' || v_suc.id_sucursal ||' - ' || v_suc.direccion);
        v_subtotal := 0; -- DENTRO del externo: vuelve a cero en cada sucursal
        v_cuenta := 0;
       
        -- ciclo interno: el valor del maestro viaja como parametro
        FOR v_cre IN c_creditos(v_suc.id_sucursal) LOOP
        DBMS_OUTPUT.PUT_LINE(' Solicitud ' || v_cre.nro_solic_credito ||' ' || TO_CHAR(v_cre.monto_credito, '999G999G999') ||' ' || v_cre.total_cuotas_credito || ' cuotas');

        v_subtotal := v_subtotal + v_cre.monto_credito; -- nivel interno
        v_total := v_total + v_cre.monto_credito; -- nivel externo
        v_cuenta := v_cuenta + 1;
        END LOOP;
        DBMS_OUTPUT.PUT_LINE(' Subtotal sucursal: ' ||TO_CHAR(v_subtotal, '999G999G999') ||' (' || v_cuenta || ' creditos)');
        DBMS_OUTPUT.PUT_LINE(' ');
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('TOTAL GENERAL: ' || TO_CHAR(v_total, '999G999G999'));
END;
```
