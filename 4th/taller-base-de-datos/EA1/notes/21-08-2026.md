## Cursor 

```sql
DECLARE
-- 1. DECLARAR: se define la consulta, no se ejecuta.
    CURSOR c_creditos IS
        SELECT  nro_solic_credito, 
                monto_credito, 
                total_cuotas_credito
        FROM CREDITO_CLIENTE
        ORDER BY nro_solic_credito;
        
-- La variable de fila copia la estructura del cursor.
v_fila c_creditos%ROWTYPE;

BEGIN
    OPEN c_creditos; -- 2. ABRIR
        LOOP
        FETCH c_creditos INTO v_fila; -- 3. RECUPERAR
            EXIT WHEN c_creditos%NOTFOUND; -- evaluar DESPUES del fetch
            DBMS_OUTPUT.PUT_LINE(v_fila.nro_solic_credito || ' -> ' ||v_fila.monto_credito);
        END LOOP;
    CLOSE c_creditos; -- 4. CERRAR
END;
/

```

## Cursor con for

```sql
DECLARE
    CURSOR c_creditos IS
        SELECT  nro_solic_credito, 
                monto_credito
        FROM CREDITO_CLIENTE
        ORDER BY nro_solic_credito;
BEGIN
    FOR v_fila IN c_creditos LOOP
        DBMS_OUTPUT.PUT_LINE(v_fila.nro_solic_credito || ' -> ' ||v_fila.monto_credito);
    END LOOP;
END;
/
```

## Atributos ddel cursor

```sql
DECLARE
    CURSOR c_creditos IS SELECT nro_solic_credito FROM CREDITO_CLIENTE;
    v_fila c_creditos%ROWTYPE;
    v_total PLS_INTEGER;
BEGIN
    OPEN c_creditos;
        LOOP
            FETCH c_creditos INTO v_fila;
            EXIT WHEN c_creditos%NOTFOUND;
            DBMS_OUTPUT.PUT_LINE(c_creditos%ROWCOUNT || '. ' || v_fila.nro_solic_credito);
        END LOOP;
    v_total := c_creditos%ROWCOUNT; -- se guarda ANTES de cerrar
    CLOSE c_creditos;
    IF v_total = 0 THEN
    DBMS_OUTPUT.PUT_LINE('No se encontraron creditos.');
    ELSE
    DBMS_OUTPUT.PUT_LINE('Total procesados: ' || v_total);
    END IF;
END;
/
```

## Cursores con parametros

```sql
DECLARE
    -- Los parametros se declaran aqui, sin tamano en el tipo.
    CURSOR c_por_sucursal (p_sucursal NUMBER) IS
        SELECT nro_solic_credito, monto_credito
        FROM CREDITO_CLIENTE
        WHERE id_sucursal = p_sucursal
        ORDER BY nro_solic_credito;
BEGIN
    -- El valor se entrega aqui, al abrir.
    FOR v_fila IN c_por_sucursal(2011) LOOP
    DBMS_OUTPUT.PUT_LINE(v_fila.nro_solic_credito);
    END LOOP;
    
    -- El mismo cursor, otro valor.
    FOR v_fila IN c_por_sucursal(4011) LOOP
    DBMS_OUTPUT.PUT_LINE(v_fila.nro_solic_credito);
    END LOOP;
END;
/
```
