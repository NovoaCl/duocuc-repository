```sql
DECLARE

    TYPE t_comuna IS RECORD(
        v_cod_region        comuna.cod_region%TYPE,
        v_cod_provincia     comuna.cod_provincia%TYPE,
        v_cod_comuna        comuna.cod_comuna%TYPE,
        v_nombre_comuna     comuna.nombre_comuna%TYPE
    );

    v_comuna      t_comuna;
    v_comuna_2    t_comuna;

BEGIN

    SELECT * INTO v_comuna FROM comuna
    WHERE cod_region = 1
    AND cod_provincia = 2
    AND cod_comuna = 3
    ;

DBMS_OUTPUT.put_line('nombre_comuna '||v_comuna.v_nombre_comuna);
--DBMS_OUTPUT.put_line('nombre_comuna '||v_comuna_2.v_nombre_comuna);
DBMS_OUTPUT.put_line('fin');
END;
```

## VARRAY

```sql

DECLARE
    TYPE t_tasas IS VARRAY(5) OF NUMBER(4,2);
    v_tasas t_tasas := t_tasas(0.50, 0.25, 0.40, 0.35, 0.80);
BEGIN
    FOR I IN 1 .. v_tasas.COUNT LOOP
        DBMS_OUTPUT.PUT_LINE('Producto ' || i || ':' || TO_CHAR(v_tasas(i), 'FM0D999') || '%');
    END LOOP;
END;
/
```
